/*
 * Yourself - Supabase Edge Function: delete-account
 * Native Deno + fetch only. No npm packages or Supabase SDK.
 * Deploy this file as the `delete-account` Edge Function.
 */

const corsHeaders = {
  'Access-Control-Allow-Origin': '*',
  'Access-Control-Allow-Headers': 'authorization, x-client-info, apikey, content-type',
  'Access-Control-Allow-Methods': 'POST, OPTIONS',
  'Content-Type': 'application/json'
};

function reply(status: number, body: Record<string, unknown>) {
  return new Response(JSON.stringify(body), { status, headers: corsHeaders });
}

async function readJson(response: Response) {
  const text = await response.text();
  if (!text) return null;
  try { return JSON.parse(text); }
  catch { return { message: text }; }
}

Deno.serve(async (request: Request) => {
  if (request.method === 'OPTIONS') return new Response('ok', { headers: corsHeaders });
  if (request.method !== 'POST') return reply(405, { error: 'method_not_allowed' });

  const baseUrl = Deno.env.get('SUPABASE_URL');
  const publicKey = Deno.env.get('SUPABASE_ANON_KEY') || Deno.env.get('SUPABASE_PUBLISHABLE_KEY');
  const serviceKey = Deno.env.get('SUPABASE_SERVICE_ROLE_KEY');
  const authorization = request.headers.get('Authorization') || '';

  if (!baseUrl || !publicKey || !serviceKey || !authorization.startsWith('Bearer ')) {
    return reply(503, { error: 'function_not_configured' });
  }

  let body: { target_user_id?: unknown };
  try { body = await request.json(); }
  catch { return reply(400, { error: 'invalid_json' }); }

  // Verify the caller's access token with Supabase Auth.
  const callerResponse = await fetch(`${baseUrl}/auth/v1/user`, {
    headers: { apikey: publicKey, Authorization: authorization }
  });
  const caller = await readJson(callerResponse);
  if (!callerResponse.ok || !caller?.id) return reply(401, { error: 'unauthorized' });

  const targetId = typeof body.target_user_id === 'string' ? body.target_user_id : caller.id;
  const uuidPattern = /^[0-9a-f]{8}-[0-9a-f]{4}-[1-8][0-9a-f]{3}-[89ab][0-9a-f]{3}-[0-9a-f]{12}$/i;
  if (!uuidPattern.test(targetId)) return reply(400, { error: 'invalid_target' });

  // A user may delete their own account. Deleting someone else requires developer role.
  if (targetId !== caller.id) {
    const roleQuery = new URLSearchParams({
      select: 'user_id',
      user_id: `eq.${caller.id}`,
      role: 'eq.developer',
      limit: '1'
    });
    const roleResponse = await fetch(`${baseUrl}/rest/v1/app_user_roles?${roleQuery}`, {
      headers: { apikey: serviceKey, Authorization: `Bearer ${serviceKey}` }
    });
    const roles = await readJson(roleResponse);
    if (!roleResponse.ok || !Array.isArray(roles) || roles.length === 0) {
      return reply(403, { error: 'not_authorized' });
    }
  }

  // List and remove the target user's private avatar files.
  for (let page = 0; page < 100; page += 1) {
    const listResponse = await fetch(`${baseUrl}/storage/v1/object/list/avatars`, {
      method: 'POST',
      headers: {
        apikey: serviceKey,
        Authorization: `Bearer ${serviceKey}`,
        'Content-Type': 'application/json'
      },
      body: JSON.stringify({ prefix: targetId, limit: 100, offset: 0, sortBy: { column: 'name', order: 'asc' } })
    });
    const files = await readJson(listResponse);
    if (!listResponse.ok || !Array.isArray(files)) {
      return reply(500, { error: 'avatar_cleanup_failed' });
    }
    const paths = files.filter((file: any) => file && file.id && typeof file.name === 'string')
      .map((file: any) => `${targetId}/${file.name}`);
    if (paths.length === 0) break;

    const deleteFilesResponse = await fetch(`${baseUrl}/storage/v1/object/avatars`, {
      method: 'DELETE',
      headers: {
        apikey: serviceKey,
        Authorization: `Bearer ${serviceKey}`,
        'Content-Type': 'application/json'
      },
      body: JSON.stringify({ prefixes: paths })
    });
    if (!deleteFilesResponse.ok) return reply(500, { error: 'avatar_cleanup_failed' });
  }

  // Delete the Auth user. Foreign-key cascades remove profile, roles and app data.
  const deleteUserResponse = await fetch(`${baseUrl}/auth/v1/admin/users/${targetId}`, {
    method: 'DELETE',
    headers: { apikey: serviceKey, Authorization: `Bearer ${serviceKey}` }
  });
  if (!deleteUserResponse.ok) {
    const details = await readJson(deleteUserResponse);
    return reply(400, { error: 'delete_failed', details });
  }

  return reply(200, { success: true });
});
